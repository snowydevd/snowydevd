function sprCard(ev) {                                       // contenido de una tarjeta (940×220, fondo transparente)
  const s = mk(940, 220), x = s.x; x.textBaseline = 'alphabetic';
  x.fillStyle = YEL; rr(x, 20, 20, 152, 180, 14); x.fill();
  x.fillStyle = BG; x.textAlign = 'center'; x.font = FD(46); x.fillText(ev.dow, 96, 70); x.font = FD(116); x.fillText(ev.day, 96, 180);
  let fs = 82; while (tw(ev.name, FD(fs)) > 650) fs -= 2;
  x.textAlign = 'left'; x.fillStyle = K.k9; x.font = FD(fs); x.fillText(ev.name, 204, 106);
  x.font = FU(41); x.fillText(ev.time, 204, 170);
  x.fillStyle = K.k7; x.fillText('·  ' + ev.zone, 204 + tw(ev.time + ' ', FU(41)) + 6, 170);
  x.strokeStyle = K.k6; x.lineWidth = 5; x.lineCap = 'round'; x.lineJoin = 'round'; x.beginPath(); x.moveTo(886, 92); x.lineTo(906, 110); x.lineTo(886, 128); x.stroke();
  return s;
}

}
function cardBox(c, w, h, hot) {
  rr(c, 0, 0, w, h, 22); c.fillStyle = K.k2; c.fill(); c.lineWidth = 2; c.strokeStyle = K.k4; c.stroke();
  if (hot > 0.01) { const a = c.globalAlpha; c.globalAlpha = a * Math.min(1, hot); c.lineWidth = 3; c.strokeStyle = YEL; c.stroke(); c.globalAlpha = a; }
}
